# Los verbos que usa White Sands, en el lenguaje de docs/LENGUAJE.md (§6).
# Primitivos (los hace el mundo), compuestos por receta (fijos), por búsqueda (A* sobre el estado) y por mapa (A* sobre
# el mapa). El primer parámetro es siempre quien lo hace. Las etiquetas de los pasos son los momentos a los que se
# engancha una escena: «ida.camino.playa cumplido» es el paso «playa» del paso «camino» del encargo «ida».

# ---------------------------------------------------------------- primitivos

=== subir(quien: persona, medio: medio) ===
primitivo
pide: mundo.quien.lugar = mundo.medio.lugar, mundo.quien.en = nada
deja: mundo.quien.en = medio
dura: 10s

=== bajar(quien: persona) ===
primitivo
pide: mundo.quien.en != nada
deja: mundo.quien.en = nada
dura: 10s

=== enganchar(quien: persona, medio: medio) ===
primitivo
pide: mundo.quien.lugar = mundo.medio.lugar
deja: mundo.medio.listo = si
dura: 5m

=== tomar_de_la_mano(quien: persona, otro: persona) ===
primitivo
pide: mundo.quien.lugar = mundo.otro.lugar
deja: mundo.otro.sigue = quien
dura: 5s

=== soltar(quien: persona, otro: persona) ===
primitivo
pide: mundo.otro.sigue = quien
deja: mundo.otro.sigue = nadie
dura: 1s

=== esperar(quien: persona, hasta: condición) ===
primitivo
# (no deja nada: se cumple cuando se cumple «hasta»)

# ---------------------------------------------------------------- por mapa

=== ir_a(quien: persona, a: lugar) ===
mapa
# (si quien va en un medio, el medio va con él y con todos los que lleva; quien la sigue, también)

=== correr_a(quien: persona, a: lugar) ===
mapa
paso: rápido
# (deja el medio en que iba: si va en la calesa, primero salta)

# ---------------------------------------------------------------- por receta

=== traer(quien: persona, otro: persona, a: lugar) ===
receta:
  va:     ir_a(quien, mundo.otro.lugar)
  toma:   tomar_de_la_mano(quien, otro)
  vuelve: ir_a(quien, a)
  suelta: soltar(quien, otro)

=== esperar_en(quien: persona, lugar: lugar, hasta: condición) ===
receta:
  va:     ir_a(quien, lugar)
  espera: esperar(quien, hasta)

=== viajar(quien: persona, medio: medio, por: ruta, a: lugar) ===
pide: mundo.quien.en = medio
receta:
  por cada parada en tramo(por, mundo.quien.lugar, a): ir_a(quien, parada)
# (cada vuelta toma la etiqueta de su parada: «camino.hondonada_lynde», «camino.escuela»…)

=== llevar_a(quien: persona, otro: persona, a: lugar, medio: medio, por: ruta) ===
receta:
  engancha: enganchar(quien, medio)
  sube:     conseguir_que(quien, otro, mundo.otro.en = medio)
  arriba:   subir(quien, medio)
  camino:   viajar(quien, medio, por, a)
  baja:     bajar(otro), bajar(quien)
mantiene: mundo.otro.en = medio desde camino, si se rompe vuelve a sube
cumplido si: mundo.otro.lugar = a
# (si Ana salta de la calesa en el camino, se rompe lo que mantiene: Marilla vuelve a «sube» —consigue que vuelva— y
# el camino sigue desde la parada en que iban)

# ---------------------------------------------------------------- por búsqueda

=== conseguir_que(quien: persona, otro: persona, c: condición) ===
busca: c
con: pedir, ordenar, traer, esperar
intentos: 4
# (el A* elige el plan más barato: pedírselo si Ana la oye; ordenárselo si ya se lo pidió; ir a buscarla y traerla si
# está lejos o no obedece. Cada intento fallido —la jugadora no obedece— vuelve a planificar; al cuarto, falla)
