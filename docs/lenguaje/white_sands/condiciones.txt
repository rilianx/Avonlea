# Las condiciones con nombre de White Sands, en el lenguaje de docs/LENGUAJE.md (§5.3).
# Se escriben una vez y se usan como cualquier condición.

# Ana está a prueba: todavía no la decidieron
a_prueba := estado.ana_en_tv < decidida

# a Marilla todavía le queda paciencia: habla en el camino; si no, el viaje es en silencio y duda de verdad
marilla_con_paciencia := cualidad.marilla.paciencia.ana >= 1

# es la hora del viaje: después del almuerzo, antes de que oscurezca
hora_de_salir := tiempo.hora en 14..19
